#include <stdio.h>
#include <stdlib.h>

#define MAX_NR 1000
#define MAX_NC 1000

// Struttura che rappresenta una cella della griglia
struct cella {
    int r;
    int c;
};

// Struttura che rappresenta una mossa
struct mossa {
    int dr;
    int dc;
};

// Funzione che legge la griglia dal file
int leggi_griglia(char* nome_file, int M[][MAX_NC], int* nr, int* nc) {
    FILE* f;
    int i, j;
    f = fopen(nome_file, "r");
    if (f == NULL) {
        return -1;
    }
    if (fscanf(f, "%d %d", nr, nc) != 2) {
        return -1;
    }
    for (i = 0; i < *nr; i++) {
        for (j = 0; j < *nc; j++) {
            if (fscanf(f, "%d", &M[i][j]) != 1) {
                return -1;
            }
        }
    }
    fclose(f);
    return 0;
}

// Funzione che stampa la griglia
void stampa_griglia(int M[][MAX_NC], int nr, int nc) {
    int i, j;
    for (i = 0; i < nr; i++) {
        for (j = 0; j < nc; j++) {
            printf("%d ", M[i][j]);
        }
        printf("\n");
    }
}

// Funzione che controlla se la cella (r, c) è valida (cioè se è bianca e non è fuori dalla griglia)
int cella_valida(int M[][MAX_NC], int nr, int nc, int r, int c) {
    if (r < 0 || r >= nr || c < 0 || c >= nc) {
        return 0; // fuori dalla griglia
    }
    if (M[r][c] == 1) {
        return 0; // cella nera
    }
    return 1; // cella bianca
}

// Funzione che calcola il numero di curve presenti nel percorso
int conta_curve(struct cella* percorso, int n) {
    int i, curve = 0;
    if (n < 2) {
        return 0;
    }
    for (i = 1; i < n; i++) {
        if (percorso[i].r - percorso[i-1].r != percorso[i-1].r - percorso[i-2].r ||
            percorso[i].c - percorso[i-1].c != percorso[i-1].c - percorso[i-2].c) {
            curve++;
        }
    }
    return curve;
}
