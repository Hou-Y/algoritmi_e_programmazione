square **readgrid(FILE *in, int *nr, int *nc){
	int i,j;
	int *tmp;
	fscanf(in, "%d %d" , nr, nc);
	tmp=malloc(*nr* (*nc)*sizeof(int));
	square **griglia=malloc(*nr*sizeof(square*);
	for(i=0; i<*nr; i++){
	*griglia=calloc(*nc*sizeof(square)); //per azzerare numero di sovrapposizioni
	}
	fgets(in, nr*nc, );
	for(i=0; i<nr; i++){
	for(j=0; j<nc; j++)
	sscanf(tmp, "%d", &(griglia[i][j].val));
	}
	return griglia;
}

				if(M[k][ris[i].corj].sovrap>2) return 0;}
			if(ris[i].direz==vert){
				for(k=ris[i].cornj; k<ris[i].cornj+ris[i].lung; k++){
				M[ris[i].corni][k].sovrap+=1;
				}
				if(M[i][corni][k].sovrap>2) return 0;}
			if(ris[i].direz==obliquo){
				for(k=0; k<ris[i].lung; k++){
				M[k+ris[i].corni][k+ris[i].cornj]+=1;
				}
				if(M[k+ris[i].corni][k+ris[i].cornj]>2) return 0;}
		}
