int ** getGriglia(int *NR, int *NC, FILE *f){
int i, j;
int **M;

fscanf(f, "%d %d", NR, NC);
M=malloc2d(NR, NC, -1);
for(i=0; i<*NR; i++){
	for(j=0; j<*NC; j++)
		fscanf(in, "%d", &M[i][j]);
	}
}
return M;
}
